 123      out = []

 125          df = pd.concat([df, event_features(df, window_h)], axis=1)

 178  REARM_W = 15.0   # строк `rearm` 2 % выборки, подтверждений у них 37 % против 3 % у тиков
 179  
 180  
 181  def sample_weights(rows: pd.DataFrame) -> np.ndarray:
 182      """Вес строк `rearm` поднят до `REARM_W`, у тиков остаётся единица.
 183  
 184      Деревья с листом от 200 строк ради 2 % выборки по событийным признакам не
 185      делятся. Отдельная модель `rearm` в серии 1 проиграла (тикам нужны эти строки),
 186      поэтому вес, а не разделение выборки.
 187      """
 188      heavy = np.where((rows["kind"] == "rearm").to_numpy(), REARM_W, 1.0)
 189      return rows["w"].to_numpy(dtype=float) * heavy
